/** * Anniversary Registration — Google Sheets backend * Paste this entire file into Extensions > Apps Script for your Sheet, * then deploy as a Web App (Execute as: Me, Who has access: Anyone). */ const REG_SHEET_NAME = 'Registrations'; const CONFIG_SHEET_NAME = 'Config'; const DEFAULT_CONFIG = { church: 'Jesus Christ To God Be The Glory Singapore Extension Church', number: '24', date: 'Sunday, 04 October 2026', time: '2:00 PM', venue: '42 Mactaggart Building, Mactaggart Road Unit #03-01 Singapore 368086', passcode: 'anniversary2026' }; function getRegSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(REG_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(REG_SHEET_NAME); sheet.appendRow(['code', 'name', 'email', 'phone', 'relatives', 'notes', 'checkedIn', 'registeredAt']); } return sheet; } function getConfigSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(CONFIG_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG_SHEET_NAME); sheet.appendRow(['church', 'number', 'date', 'time', 'venue', 'passcode']); // Force date/time columns to plain text so Sheets doesn't auto-convert // strings like "Sunday, 04 October 2026" into real date serial values. sheet.getRange(2, 3, 1, 2).setNumberFormat('@'); sheet.appendRow([ DEFAULT_CONFIG.church, DEFAULT_CONFIG.number, DEFAULT_CONFIG.date, DEFAULT_CONFIG.time, DEFAULT_CONFIG.venue, DEFAULT_CONFIG.passcode ]); } return sheet; } function cellToText_(val, isTime) { // Safety net: if a cell still holds a real Date/Time object instead of // plain text, convert it back to a readable string. Google Sheets stores // time-only values using a base date of Dec 30, 1899 — detect that and // format as a clock time instead of a calendar date. if (Object.prototype.toString.call(val) === '[object Date]') { const tz = Session.getScriptTimeZone(); if (isTime || val.getFullYear() === 1899) { return Utilities.formatDate(val, tz, 'h:mm a'); } return Utilities.formatDate(val, tz, "EEEE, dd MMMM yyyy"); } return val; } function readConfig_() { const sheet = getConfigSheet_(); const row = sheet.getRange(2, 1, 1, 6).getValues()[0]; return { church: row[0] || DEFAULT_CONFIG.church, number: row[1] || DEFAULT_CONFIG.number, date: cellToText_(row[2], false) || DEFAULT_CONFIG.date, time: cellToText_(row[3], true) || DEFAULT_CONFIG.time, venue: row[4] || DEFAULT_CONFIG.venue, passcode: row[5] || DEFAULT_CONFIG.passcode }; } function writeConfig_(cfg) { const sheet = getConfigSheet_(); sheet.getRange(2, 3, 1, 2).setNumberFormat('@'); sheet.getRange(2, 1, 1, 6).setValues([[ cfg.church, cfg.number, cfg.date, cfg.time, cfg.venue, cfg.passcode ]]); } function readAllRegistrations_() { const sheet = getRegSheet_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; const rows = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); return rows .filter(r => r[0]) // skip blank rows .map(r => ({ code: r[0], name: r[1], email: r[2], phone: r[3], relatives: r[4] ? JSON.parse(r[4]) : [], notes: r[5], checkedIn: r[6] === true || r[6] === 'TRUE', registeredAt: r[7] })); } function findRowByCode_(sheet, code) { const lastRow = sheet.getLastRow(); if (lastRow < 2) return -1; const codes = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < codes.length; i++) { if (codes[i][0] === code) return i + 2; // 1-indexed, +1 for header } return -1; } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const action = e.parameter.action; try { if (action === 'list') { return jsonOut_({ ok: true, registrations: readAllRegistrations_() }); } if (action === 'config') { return jsonOut_({ ok: true, config: readConfig_() }); } return jsonOut_({ ok: false, error: 'Unknown action' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } function doPost(e) { let data; try { data = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ ok: false, error: 'Invalid request body' }); } const action = data.action; const cfg = readConfig_(); try { if (action === 'register') { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sheet = getRegSheet_(); // Sheet has a header row, so getLastRow() already equals // (existing entries + 1) — i.e. the next sequential entry number. const nextNum = sheet.getLastRow(); const code = '#' + nextNum; const now = new Date().toISOString(); sheet.appendRow([ code, data.name || '', data.email || '', data.phone || '', JSON.stringify(data.relatives || []), '', false, now ]); return jsonOut_({ ok: true, code: code, registeredAt: now }); } finally { lock.releaseLock(); } } if (action === 'checkin') { if (data.passcode !== cfg.passcode) { return jsonOut_({ ok: false, error: 'Invalid passcode' }); } const sheet = getRegSheet_(); const rowIndex = findRowByCode_(sheet, data.code); if (rowIndex === -1) return jsonOut_({ ok: false, error: 'Registration not found' }); const current = sheet.getRange(rowIndex, 7).getValue(); const next = data.hasOwnProperty('checkedIn') ? !!data.checkedIn : !current; sheet.getRange(rowIndex, 7).setValue(next); return jsonOut_({ ok: true, checkedIn: next }); } if (action === 'saveConfig') { if (data.passcode !== cfg.passcode) { return jsonOut_({ ok: false, error: 'Invalid passcode' }); } const newConfig = { church: data.church || cfg.church, number: data.number || cfg.number, date: data.date || cfg.date, time: data.time || cfg.time, venue: data.venue || cfg.venue, passcode: data.newPasscode || cfg.passcode }; writeConfig_(newConfig); return jsonOut_({ ok: true, config: newConfig }); } if (action === 'clearAll') { if (data.passcode !== cfg.passcode) { return jsonOut_({ ok: false, error: 'Invalid passcode' }); } const sheet = getRegSheet_(); const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 8).clearContent(); } return jsonOut_({ ok: true }); } return jsonOut_({ ok: false, error: 'Unknown action' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } }
